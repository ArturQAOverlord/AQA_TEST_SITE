Пройдись по карточкам, достань текущую цену из элемента с data-testid="price" и переведи её в число:

cards = page.get_by_role("listitem")
prices = [int(re.sub(r"\D", "", card.get_by_test_id("price").inner_text())) for card in cards.all()]
cards.nth(prices.index(min(prices))).get_by_role("button", name="Купить").click()
